VERIFICACION (corregida): el Core NO confia en el hash del minero.
El Core recalcula la tarea el mismo (trustless) y compara el resultado

def gen_matriz(n: int, seed: int) -> np.ndarray:
    """Matriz determinista de enteros acotados (exacta en float64)."""
    idx = np.arange(n, dtype=np.float64)
    return (idx[:, None] * 31 + idx[None, :] * 17 + seed) % 97 - 48

def computar_tarea(tipo: str, params: dict):
    """Recalcula el resultado esperado en el propio Core (numpy, CPU)."""
    if tipo == "checksum":
        data = str(params.get("data", "kukia"))
        return hashlib.sha256(data.encode()).hexdigest()
    if tipo == "matriz":
        n = int(params.get("n", 256))
        seed = int(params.get("seed", 42))
        a = gen_matriz(n, seed)
        b = gen_matriz(n, seed + 1)
        return str(int(round(float((a @ b).sum()))))
    if tipo == "inferencia":
        n = int(params.get("n", 64))
        seed = int(params.get("seed", 7))
        x = gen_matriz(n, seed)
        k = np.array([[1.0, 2.0, 1.0], [2.0, 4.0, 2.0], [1.0, 2.0, 1.0]])
        out = 0.0
        for i in range(3):
            for j in range(3):
                out += float((x[i:n-2+i, j:n-2+j] * k[i, j]).sum())
        return str(int(round(out)))
    return None

        return {"tarea": None}
    con.execute("UPDATE tareas SET estado='ejecutando', nodo_id=? WHERE id=?", (nodo_id, row["id"]))

@app.post("/tareas/resultado")
def resultado(tid: str, nodo_id: str, metrica: str, hash_res: str = ""):
    """El Core recalcula la tarea y compara el RESULTADO EXACTO (trustless)."""
    task = con.execute("SELECT * FROM tareas WHERE id=?", (tid,)).fetchone()
    if not task:
        raise HTTPException(404, "tarea no encontrada")
    params = json.loads(task["params"])
    esperado = computar_tarea(task["tipo"], params)
    if esperado is None:
        raise HTTPException(400, "tipo de tarea no soportado")
    ok = str(metrica).strip() == str(esperado).strip()
    kuki = 0.0
    if ok:
        node = con.execute("SELECT * FROM nodos WHERE id=?", (nodo_id,)).fetchone()
        if not node:
            raise HTTPException(404, "nodo no registrado")
        kuki = REWARD_BASE * node["tier"]
        con.execute("UPDATE nodos SET tareas_ok=tareas_ok+1, reputacion=min(5,reputacion+0.05) "
                    "WHERE id=?", (nodo_id,))
        h = hashlib.sha256(f"{tid}:{nodo_id}:{metrica}".encode()).hexdigest()
        con.execute("UPDATE tareas SET estado='ok', hash=?, metrica=?, fin=?, verificado=1 WHERE id=?",
                    (h, metrica, time.ctime(), tid))
        con.execute("INSERT INTO ledger(tarea_id,nodo_id,kuki,motivo,ts) VALUES(?,?,?,?,?)",
                    (tid, nodo_id, kuki, "trabajo verificado", time.ctime()))
    else:
        con.execute("UPDATE tareas SET estado='fallo', fin=? WHERE id=?", (time.ctime(), tid))
        con.execute("INSERT INTO ledger(tarea_id,nodo_id,kuki,motivo,ts) VALUES(?,?,?,?,?)",
                    (tid, nodo_id, 0.0, "resultado incorrecto (recalculado por el Core)", time.ctime()))
    con.commit()
    return {"ok": ok, "kuki": kuki, "esperado": esperado}

"""KUKIA Worker - minero multi-GPU: se registra, pide trabajo, ejecuta en GPU.

VERIFICACION REAL: el worker calcula el resultado (no un hash propio) y el
Core lo recalcula y compara. El worker NO puede falsificar: si el numero no
coincide exactamente con el recalculado por el Core, no cobra.

MULTI-GPU: usa TODAS las GPUs disponibles a la vez (una cola por GPU).
"""
import hashlib, json, sys, time, threading, socket
import requests

CORE = "http://127.0.0.1:8300"

def trabajo_local(tipo: str, params: dict, device: str = "cuda:0") -> str:
    """Calcula la tarea EXACTAMENTE como el Core (determinista)."""
    if tipo == "matriz":
        import torch
        n = int(params.get("n", 256))
        seed = int(params.get("seed", 42))
        idx = torch.arange(n, device=device, dtype=torch.float64)
        a = ((idx[:, None] * 31 + idx[None, :] * 17 + seed) % 97) - 48
        b = ((idx[:, None] * 31 + idx[None, :] * 17 + (seed + 1)) % 97) - 48
        c = float((a @ b).sum().item())
        return str(int(round(c)))
    if tipo == "checksum":
        data = str(params.get("data", "kukia"))
        return hashlib.sha256(data.encode()).hexdigest()
    if tipo == "inferencia":
        import torch
        n = int(params.get("n", 64))
        seed = int(params.get("seed", 7))
        idx = torch.arange(n, device=device, dtype=torch.float64)
        x = ((idx[:, None] * 31 + idx[None, :] * 17 + seed) % 97) - 48
        k = torch.tensor([[1.0, 2.0, 1.0], [2.0, 4.0, 1.0 * 2], [1.0, 2.0, 1.0]],
                         device=device, dtype=torch.float64)
        out = 0.0
        for i in range(3):
            for j in range(3):
                out += float((x[i:n-2+i, j:n-2+j] * k[i, j]).sum().item())
        return str(int(round(out)))
    return "tarea_no_soportada"

def ciclo_worker(nombre: str, device: str, vram_gb: int, tops: float):
    """Un hilo por GPU: registra su nodo y mina en bucle."""
    etiqueta = f"{nombre}[{device}]"
    try:
        r = requests.post(f"{CORE}/nodos/registrar",
                          json={"nombre": etiqueta, "vram_gb": vram_gb, "tops": tops}, timeout=30)
        nid = r.json()["id"]
        print(f"[KUKIA] nodo {nid} ({etiqueta}) vram={vram_gb}GB listo")
    except Exception as e:
        print(f"[KUKIA] {etiqueta} error registro: {e}")
        return
    while True:
        try:
            pend = requests.get(f"{CORE}/tareas/pendiente", params={"nodo_id": nid}, timeout=30).json()["tarea"]
            if not pend:
                time.sleep(3)
                continue
            metrica = trabajo_local(pend["tipo"], pend["params"], device)
            res = requests.post(f"{CORE}/tareas/resultado",
                                params={"tid": pend["id"], "nodo_id": nid, "metrica": metrica},
                                timeout=60).json()
            estado = f"OK +{res['kuki']} KIA" if res.get("ok") else f"RECHAZADO (esperado {res.get('esperado')})"
            print(f"[KUKIA] {etiqueta} tarea {pend['id'][:8]} -> {estado}")
        except Exception as e:
            print(f"[KUKIA] {etiqueta} err: {e}")
            time.sleep(5)

def main():
    nombre = sys.argv[1] if len(sys.argv) > 1 else "worker_" + socket.gethostname()
    try:
        import torch
        n_gpus = torch.cuda.device_count()
    except Exception:
        n_gpus = 0

    if n_gpus == 0:
        print("[KUKIA] Sin GPUs CUDA - usando CPU (mas lento)")
        ciclo_worker(nombre, "cpu", 0, 0.0)
        return

    print(f"[KUKIA] {n_gpus} GPU(s) detectada(s) - minando en TODAS a la vez")
    hilos = []
    for g in range(n_gpus):
        try:
            props = torch.cuda.get_device_properties(g)
            vram = props.total_memory // (1024**3)
            # TOPS aproximados por clase de GPU (referencia NVIDIA)
            tops_est = {8: 60.0, 12: 300.0, 16: 760.0, 24: 1300.0}.get(int(vram), 100.0)
        except Exception:
            vram, tops_est = 0, 0.0
        t = threading.Thread(target=ciclo_worker, args=(nombre, f"cuda:{g}", vram, tops_est), daemon=True)
        t.start()
        hilos.append(t)
    for t in hilos:
        t.join()

if __name__ == "__main__":
    main()

Billetera - kukia_wallet.py

"""KUKIA Wallet - billetera con criptografia REAL (Ed25519).

- Clave privada Ed25519 (256 bits) generada con secrets (CSPRNG del sistema)
- Direccion derivada de la clave PUBLICA (no de la privada)
- Firma digital Ed25519 verificable (no HMAC simplificado)
- La clave privada nunca sale del equipo del usuario
"""
import json, hashlib, os, secrets, time
from pathlib import Path

WALLET_DIR = Path(__file__).parent.parent / "ledger" / "wallet"
WALLET_DIR.mkdir(parents=True, exist_ok=True)

try:
    from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey, Ed25519PublicKey
    from cryptography.hazmat.primitives import serialization
    CRYPTO_REAL = True
except ImportError:
    CRYPTO_REAL = False

def crear_billetera(nombre):
    """Crea o devuelve la billetera del nodo (Ed25519 + saldo local)."""
    f = WALLET_DIR / f"{nombre}.json"
    if f.exists():
        d = json.loads(f.read_text(encoding="utf-8"))
        return d

    if CRYPTO_REAL:
        # clave privada Ed25519 real (CSPRNG del sistema)
        priv = Ed25519PrivateKey.generate()
        priv_bytes = priv.private_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PrivateFormat.Raw,
            encryption_algorithm=serialization.NoEncryption())
        pub_bytes = priv.public_key().public_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PublicFormat.Raw)
        clave_secreta = priv_bytes.hex()
        clave_publica = pub_bytes.hex()
        # direccion = hash de la clave PUBLICA (estandar cripto)
        direccion = "KUK" + hashlib.sha256(pub_bytes).hexdigest()[:38].upper()
        esquema = "ed25519"
    else:
        # fallback (no deberia ocurrir: cryptography esta instalado)
        clave_secreta = secrets.token_hex(32)
        clave_publica = ""
        direccion = "KUK" + hashlib.sha256(clave_secreta.encode()).hexdigest()[:38].upper()
        esquema = "sha256-fallback"

    d = {"nombre": nombre, "direccion": direccion, "esquema": esquema,
         "clave_privada": clave_secreta, "clave_publica": clave_publica,
         "saldo": 0.0, "creada": time.ctime()}
    f.write_text(json.dumps(d, indent=2), encoding="utf-8")
    return d

def firmar(datos, clave_privada_hex):
    """Firma un mensaje con la clave privada Ed25519 (firma digital real)."""
    if not CRYPTO_REAL:
        return hashlib.sha256(f"{datos}:{clave_privada_hex}".encode()).hexdigest()
    priv = Ed25519PrivateKey.from_private_bytes(bytes.fromhex(clave_privada_hex))
    return priv.sign(datos.encode()).hex()

def verificar_firma(datos, firma_hex, clave_publica_hex):
    """Verifica una firma Ed25519 contra la clave publica (True/False)."""
    if not CRYPTO_REAL:
        return False
    try:
        pub = Ed25519PublicKey.from_public_bytes(bytes.fromhex(clave_publica_hex))
        pub.verify(bytes.fromhex(firma_hex), datos.encode())
        return True
    except Exception:
        return False

def ver_saldo(nombre):
    d = crear_billetera(nombre)
    return d["saldo"]

def abonar(nombre, monto, motivo):
    d = crear_billetera(nombre)
    d["saldo"] = round(d["saldo"] + monto, 4)
    f = WALLET_DIR / f"{nombre}.json"
    f.write_text(json.dumps(d, indent=2), encoding="utf-8")
    hist = WALLET_DIR / f"{nombre}_hist.json"
    h = json.loads(hist.read_text(encoding="utf-8")) if hist.exists() else []
    h.append({"ts": time.ctime(), "monto": monto, "motivo": motivo})
    hist.write_text(json.dumps(h[-200:], indent=2), encoding="utf-8")
    return d["saldo"]

if __name__ == "__main__":
    import sys
    nombre = sys.argv[1] if len(sys.argv) > 1 else "nodo_fundador"
    w = crear_billetera(nombre)
    print(f"[KUKIA Wallet] {w['direccion']}")
    print(f"  esquema: {w['esquema']}")
    print(f"  saldo: {w['saldo']} KIA")
    # prueba de firma/verificacion
    if CRYPTO_REAL:
        firma = firmar("prueba-kukia", w["clave_privada"])
        ok = verificar_firma("prueba-kukia", firma, w["clave_publica"])
        print(f"  firma Ed25519 verificada: {ok}")
